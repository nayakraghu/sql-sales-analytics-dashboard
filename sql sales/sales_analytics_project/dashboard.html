-- Running total of revenue month by month
SELECT order_date, salesperson, revenue,
    SUM(revenue) OVER (
        PARTITION BY YEAR(order_date)
        ORDER BY order_date
        ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS running_total
FROM sales ORDER BY order_date;

-- Rank salespersons by total revenue
SELECT salesperson, SUM(revenue) AS total_revenue,
    RANK() OVER (ORDER BY SUM(revenue) DESC) AS revenue_rank,
    ROUND(SUM(revenue)*100.0/SUM(SUM(revenue)) OVER(),2) AS pct_of_total
FROM sales GROUP BY salesperson ORDER BY revenue_rank;

-- Month-over-month revenue growth using LAG()
WITH monthly AS (
    SELECT DATE_FORMAT(order_date,'%Y-%m') AS month, SUM(revenue) AS monthly_rev
    FROM sales GROUP BY month
)
SELECT month, monthly_rev,
    LAG(monthly_rev) OVER (ORDER BY month) AS prev_month,
    ROUND((monthly_rev - LAG(monthly_rev) OVER(ORDER BY month))
        / LAG(monthly_rev) OVER(ORDER BY month)*100,2) AS mom_growth_pct
FROM monthly;

-- Top products per category using DENSE_RANK()
SELECT product_name, category, SUM(revenue) AS total_revenue, COUNT(*) AS units_sold,
    DENSE_RANK() OVER (PARTITION BY category ORDER BY SUM(revenue) DESC) AS rank_in_category
FROM sales GROUP BY product_name, category
HAVING rank_in_category = 1 ORDER BY total_revenue DESC;
